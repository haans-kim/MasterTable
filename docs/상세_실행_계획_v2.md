```python
mecab = Mecab()

def extract_meaningful_words(text):
    """의미 있는 명사, 동사 추출"""
    pos_tags = mecab.pos(text)
    # NNG(일반명사), NNP(고유명사), VV(동사), VA(형용사) 추출
    meaningful = [word for word, tag in pos_tags
                 if tag in ['NNG', 'NNP', 'VV', 'VA'] and len(word) > 1]
    return meaningful
```

```python
class HybridTopicAnalyzer:
    def __init__(self):
        self.llm_client = None  # OpenAI or Claude API

    def analyze_responses(self, responses, question_context):
        """응답 분석 3단계 프로세스"""

        # Step 1: 통계 기반 초기 클러스터링
        initial_clusters = self.statistical_clustering(responses)

        # Step 2: LLM으로 주제 정제 및 명명
        refined_topics = self.llm_refine_topics(
            initial_clusters,
            question_context
        )

        # Step 3: 키워드 추출 및 설명 생성
        final_results = self.generate_final_output(
            refined_topics,
            responses
        )

        return final_results
```

```python
def create_topic_extraction_prompt(sample_responses, question_context):
    """주제 추출용 LLM 프롬프트"""
    prompt = f"""
    다음은 "{question_context}"에 대한 직원 응답들입니다.

    [응답 샘플]
    {sample_responses[:10]}

    위 응답들을 분석하여 5-10개의 주요 주제를 도출하고 다음 형식으로 정리해주세요:

    주제1: [간단명확한 주제명 2-4단어]
    키워드: [관련 키워드 3-4개]
```

#### 2.3 응답-주제 매칭 알고리즘
```python
class ResponseTopicMatcher:
    def __init__(self):
        self.vectorizer = TfidfVectorizer()

    def match_responses_to_topics(self, responses, topics):
        """각 응답을 가장 적합한 주제에 할당"""

        # 1. 주제별 대표 벡터 생성
        topic_vectors = self.create_topic_vectors(topics)

        # 2. 각 응답을 벡터화
        response_vectors = self.vectorizer.transform(responses)

        # 3. 코사인 유사도로 매칭
        assignments = []
        for resp_vec in response_vectors:
            similarities = cosine_similarity(resp_vec, topic_vectors)
            best_topic = np.argmax(similarities)
            assignments.append(best_topic)

        # 4. 빈도 계산
        topic_counts = Counter(assignments)

        return self.calculate_statistics(topic_counts, len(responses))
```

```python
class ExcelReportGenerator:
    def generate_company_report(self, company_name, analysis_results):
        """계열사별 분석 결과를 예시 형식으로 생성"""

        output_data = []

        for question_id, question_analysis in analysis_results.items():
            first_row = True

            for topic in question_analysis['topics']:
                row = {
                    '회사명': company_name if first_row else '',
                    '문항번호': question_id,
                    '주제': topic['name'],
                    '키워드': ', '.join(topic['keywords']),
                    '설명': topic['description'],
                    '빈도수': topic['frequency'],
                    '퍼센트': topic['percentage']
                }
                output_data.append(row)
                first_row = False

        return pd.DataFrame(output_data)
```

#### 3.2 품질 검증 체크리스트
```python
class QualityValidator:
    def validate_results(self, results):
        """결과 품질 검증"""
        checks = {
            '주제수': len(results['topics']) >= 5 and len(results['topics']) <= 10,
            '키워드수': all(len(t['keywords']) >= 3 for t in results['topics']),
            '설명길이': all(20 <= len(t['description']) <= 100 for t in results['topics']),
            '비율합계': abs(sum(t['percentage'] for t in results['topics']) - 100) < 1,
            '빈도합계': sum(t['frequency'] for t in results['topics']) == results['total_responses']
        }

        return all(checks.values()), checks
```

```python
def run_complete_analysis():
    # 1. 초기화
    processor = SurveyDataProcessor()
    analyzer = HybridTopicAnalyzer()
    generator = ExcelReportGenerator()
    validator = QualityValidator()

    # 2. 계열사별 분석
    all_results = []

    for company in companies:
        print(f"분석 중: {company}")

        # 데이터 준비
        company_data = processor.prepare_company_data(company)

        # 문항별 분석
        company_results = {}
        for question_id, responses in company_data.items():
            # LLM 분석
            analysis = analyzer.analyze_responses(
                responses,
                question_contexts[question_id]
            )

            # 품질 검증
            is_valid, check_results = validator.validate_results(analysis)
            if not is_valid:
                print(f"경고: {company} - {question_id} 품질 검증 실패")
                # 재분석 또는 수동 개입

            company_results[question_id] = analysis

        # 결과 저장
        df = generator.generate_company_report(company, company_results)
        all_results.append(df)

    # 3. 최종 통합 파일 생성
    final_df = pd.concat(all_results)
    final_df.to_excel('CJ그룹_주관식분석_최종.xlsx', index=False)

    return final_df
```
